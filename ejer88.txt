# Solicitamos la cantidad de números (N)
n = int(input("¿Cuántos números deseas ingresar (N)?: "))

# Inicializamos el acumulador en 0
suma = 0

# Estructura Para (For) que se repite N veces
for i in range(1, n + 1):
    numero = float(input(f"Ingresa el número {i}: "))
    
    # Acumulador: sumamos el número actual a lo que ya llevábamos acumulado
    suma += numero

# Verificamos que N no sea 0 para evitar división por cero
if n > 0:
    promedio = suma / n
    print("\n--- Resultados ---")
    print(f"La suma total es: {suma}")
    print(f"El promedio es: {promedio}")
else:
    print("La cantidad de números debe ser mayor a 0.")
