def mostrar_tablas():
    # Estructura Repetir (simulada con while True)
    while True:
        print("\n--- GENERADOR DE TABLAS DE MULTIPLICAR ---")
        numero = int(input("¿De qué número deseas ver la tabla de multiplicar?: "))
        
        print(f"\nTabla del {numero}:")
        print("-" * 15)
        
        # Estructura Para (For) para calcular del 1 al 10
        for i in range(1, 11):
            resultado = numero * i
            print(f"{numero} x {i} = {resultado}")
            
        print("-" * 15)
        
        # Preguntar si desea repetir el proceso (usando s/n)
        while True:
            respuesta = input("¿Deseas consultar otra tabla? (s/n): ").strip().lower()
            if respuesta in ("s", "si", "sí"):
                break # Sale del bucle interno y vuelve a repetir el proceso principal
            elif respuesta in ("n", "no"):
                print("¡Hasta luego!")
                return # Finaliza por completo la función/programa
            else:
                print("Opción no válida. Por favor, escribe 's' o 'n'.")

# Ejecutar el programa
mostrar_tablas()
