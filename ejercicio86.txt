pares = 0
impares = 0

print("Por favor, ingresa 10 números:")

# Estructura Para (For) que se repite 10 veces
for i in range(1, 11):
    numero = int(input(f"Número {i}: "))
    
    # Estructura Si (If) para verificar si es par o impar
    if numero % 2 == 0:
        pares += 1
    else:
        impares += 1

print("\n--- Resultados ---")
print(f"Cantidad de números pares: {pares}")
print(f"Cantidad de números impares: {impares}")
