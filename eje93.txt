def calcular_estadisticas_notas():
    suma = 0
    cantidad = 0
    mayor = None
    menor = None

    print("Ingresa las notas una por una.")

    while True:
        try:
            nota = float(input("Ingresa una nota (o un valor negativo para salir): "))

            print("Por favor, ingresa un número válido.")
            continue
            
        # Condición para finalizar el bucle
        if nota < 0:
            break
            
        # Validamos que la nota esté en un rango lógico (ej. de 0 a 10 o 0 a 100, opcional)

            print("Nota fuera de rango. Por favor, intenta de nuevo.")
            continue
            
        # Acumulador para la suma y contador de notas
        suma += nota

    print("\n--- RESULTADOS FINALES ---")
