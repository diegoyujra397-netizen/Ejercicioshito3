def analizar_numeros():
    pares = 0
    impares = 0
    positivos = 0
    negativos = 0
    
    print("--- ANALIZADOR DE NÚMEROS ---")
    print("Ingresa números. Escribe '0' para finalizar.\n")
    
    # Solicitamos el primer número para entrar al bucle
    numero = int(input("Ingresa un número (0 para salir): "))
    
    # Estructura Mientras (While) con operadores relacionales (!=)
    while numero != 0:
        
        # Operadores para positivo o negativo
        if numero > 0:
            positivos += 1
        elif numero < 0:
            negativos += 1
            
        # Operador aritmético de módulo (%) para par o impar
        if numero % 2 == 0:
            pares += 1
        else:
            impares += 1
            
        # Leer el siguiente número dentro del bucle
        numero = int(input("Ingresa otro número (0 para salir): "))

    # Resultados finales
    print("\n--- RESULTADOS FINALES ---")
    print(f"Cantidad de números pares: {pares}")
    print(f"Cantidad de números impares: {impares}")
    print(f"Cantidad de números positivos: {positivos}")
    print(f"Cantidad de números negativos: {negativos}")

# Ejecutar el programa
analizar_numeros()
